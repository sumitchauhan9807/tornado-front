import React from 'react';

function getYearCalendar(year) {
  const months = [];

  for (let month = 0; month < 12; month++) {
    const monthDate = new Date(year, month, 1);

    const monthName = monthDate.toLocaleString('en-US', {
      month: 'long',
    });

    const daysInMonth = new Date(year, month + 1, 0).getDate();

    const days = [];

    for (let day = 1; day <= daysInMonth; day++) {
      const date = new Date(year, month, day);
      const weekday = date.toLocaleString('en-US', {
        weekday: 'long',
      });
      let isWorking = true;
      if (weekday == 'Sunday') {
        isWorking = false;
      }
      days.push({
        wd: weekday,
        date: day,
        isWorking:isWorking
      });
    }

    months.push({
      name: monthName.toLowerCase(),
      days,
    });
  }

  return months;
}

export default function Calendar({ year = 2026 }) {
  const calendar = getYearCalendar(year);
  console.log(calendar, 'calendar');
  const weekDays = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];

  return (
    <div className="min-h-screen bg-gray-50 p-6">
      <div className="mx-auto max-w-7xl">
        <h1 className="mb-8 text-center text-3xl font-bold text-gray-900">{year} Calendar</h1>

        <div className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
          {calendar.map((month) => {
            // Get the weekday of the first day of the month
            const firstDay = new Date(year, new Date(`${month.name} 1, ${year}`).getMonth(), 1).getDay();

            return (
              <div key={month.name} className="overflow-hidden rounded-xl border border-gray-200 bg-white shadow-sm">
                {/* Month header */}
                <div className="border-b border-gray-200 px-4 py-3">
                  <h2 className="text-lg font-semibold capitalize text-gray-900">{month.name}</h2>
                </div>

                {/* Weekday header */}
                <div className="grid grid-cols-7 border-b border-gray-100">
                  {weekDays.map((day) => (
                    <div key={day} className="py-2 text-center text-xs font-medium text-gray-500">
                      {day}
                    </div>
                  ))}
                </div>

                {/* Calendar days */}
                <div className="grid grid-cols-7 p-2">
                  {/* Empty cells before first day */}
                  {Array.from({ length: firstDay }).map((_, index) => (
                    <div key={`empty-${index}`} className="h-10" />
                  ))}

                  {month.days.map((day) => (
                    <div key={day.date} className="flex h-10 items-center justify-center">
                      <button
                      style={{background:day.isWorking ? '' : 'red'}}
                        className={`
                          flex h-8 w-8 items-center justify-center
                          rounded-full text-sm
                          transition
                          hover:bg-gray-100
                        `}
                      >
                        {day.date}
                      </button>
                    </div>
                  ))}
                </div>
              </div>
            );
          })}
        </div>
      </div>
    </div>
  );
}
